using System;
using System.Threading.Tasks;
using System.Web.Mvc;
using OnlineVotingSystem.Models;
using OnlineVotingSystem.Services;

namespace OnlineVotingSystem.Controllers
{
    [Authorize]
    public class VotingController : Controller
    {
        private readonly IVoterVerificationService _verificationService;
        private readonly IElectionLedgerService _ledgerService;

        public VotingController(IVoterVerificationService verification, IElectionLedgerService ledger)
        {
            _verificationService = verification;
            _ledgerService = ledger;
        }

        // Cast Encrypted Vote with Aadhaar Validation
        [HttpPost]
        [ValidateAntiForgeryToken]
        public async Task<ActionResult> CastBallot(BallotSubmissionModel model)
        {
            if (!ModelState.IsValid)
                return Json(new { success = false, message = "Invalid ballot data." });

            var isEligible = await _verificationService.VerifyAadhaarEligibilityAsync(model.AadhaarToken);
            if (!isEligible)
                return Json(new { success = false, message = "Constituent already voted or unverified." });

            await _ledgerService.RecordEncryptedVoteAsync(model.CandidateId, model.ConstituencyId);
            return Json(new { success = true, message = "Ballot successfully registered in tamper-proof ledger." });
        }
    }
}
